High-throughput Layer 7 load balancer distributing incoming web traffic across redundant Amazon ECS container tasks in private subnets across multiple Availability Zones.

Security: TLS 1.3 Termination, AWS WAF rate-limiting rules
Redundancy: Multi-AZ with Health Checks & Auto-Target Registration
Monitoring: CloudWatch 5XX metrics & latency alarms
alb_main.tf 
resource "aws_lb" "main_app_alb" {
  name               = "finxserve-production-alb"
  internal           = false
  load_balancer_type = "application"
  security_groups    = [aws_security_group.alb_sg.id]
  subnets            = module.vpc.public_subnets

  enable_deletion_protection = true
  drop_invalid_header_fields = true

  tags = {
    Environment = "production"
    ManagedBy   = "Terraform"

name: DevSecOps CI/CD Security Pipeline

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main, develop ]

jobs:
  security-audit:
    name: Shift-Left Security Scans & Quality Gates
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Source Code
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      # 1. Secret Scanning with TruffleHog
      - name: TruffleHog Secret Detection
        uses: trufflesecurity/trufflehog@main
        with:
          path: ./
          base: ${{ github.event.repository.default_branch }}
          head: HEAD

      # 2. SAST Analysis with SonarQube
      - name: SonarQube Code Quality & Security Gate
        uses: sonarsource/sonarqube-scan-action@master
        env:
          SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}
          SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }}

      # 3. SCA Dependency Scanning with Snyk
      - name: Snyk Dependency Vulnerability Check
        uses: snyk/actions/node@master
        env:
          SNYK_TOKEN: ${{ secrets.SNYK_TOKEN }}
        with:
          args: --severity-threshold=high

      # 4. Infrastructure as Code (IaC) Scan with Checkov
      - name: Checkov IaC Security Scan
        uses: bridgecrewio/checkov-action@master
        with:
          framework: terraform
          soft_fail: false

      # 5. Build Docker Image & Scan with Aqua Trivy
      - name: Build Docker Container
        run: docker build -t myapp:${{ github.sha }} .

      - name: Aqua Trivy Container Vulnerability Scan
        uses: aquasecurity/trivy-action@master
        with:
          image-ref: myapp:${{ github.sha }}
          format: 'table'
          exit-code: '1'
          ignore-unfixed: true
          severity: 'CRITICAL,HIGH'

      # 6. Automated Slack Notification on Security Failure

        if: failure()
        uses: slackapi/slack-github-action@v1.24.0

                {
                  "type": "section",
                  "text": {
